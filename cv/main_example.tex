\documentclass[11pt,a4paper,sans]{moderncv}
\moderncvstyle{banking}
\moderncvcolor{blue}

% Force the name and section headings to render in moderncv blue (color1).
% Default banking leaves them black: moderncvstylebanking.sty's \colorlet
% copies (not aliases) the pre-scheme accent colour, so the name colours are
% frozen before \moderncvcolor runs. Re-let them after. \namefont is the hook
% every name-style macro routes through, so this also works on moderncv 2.3.1
% (Debian/Ubuntu apt), which has no \firstnamestyle/\lastnamestyle at all.
\renewcommand*{\namefont}{\fontsize{34}{36}\bfseries\upshape}
\colorlet{firstnamecolor}{color1}
\colorlet{lastnamecolor}{color1}
\colorlet{namecolor}{color1}
\renewcommand*{\sectionstyle}[1]{{\sectionfont\color{color1}#1}}

% ============================================================
%     ATS-SAFETY FIX: plain-text contact symbols
% ============================================================
% moderncv's default "banking" style draws the phone/email/homepage/social
% icons using FontAwesome5 glyphs (\faMobile, \faEnvelope, etc.). Under
% lualatex the PDF's ToUnicode CMap for these icon glyphs is not always
% embedded correctly, so text-extraction tools (including the parsers used
% by ATS systems) read back the icon's internal FontAwesome name instead of
% a blank glyph -- e.g. the mobile icon extracts as the literal text
% "MOBILE-ALT" glued directly onto the phone number, and the email icon
% extracts as "Envelope". This silently corrupts the contact-info line for
% every automated system that reads the CV as text, even though the icons
% render correctly when the PDF is viewed visually -- which is why the bug
% is easy to miss.
%
% Fix: override moderncv's documented symbol macros (moderncviconsawesome.sty)
% with plain text labels instead of icon glyphs. This removes the dependency
% on FontAwesome ligatures for the header entirely, so the extracted text is
% always clean regardless of engine (lualatex/pdflatex) or font backend.
\renewcommand*{\mobilephonesymbol}{Mobile:~}
\renewcommand*{\fixedphonesymbol}{Phone:~}
\renewcommand*{\faxphonesymbol}{Fax:~}
\renewcommand*{\emailsymbol}{Email:~}
\renewcommand*{\homepagesymbol}{Web:~}
\renewcommand*{\linkedinsocialsymbol}{LinkedIn:~}
\renewcommand*{\githubsocialsymbol}{GitHub:~}
\renewcommand*{\addresssymbol}{}
% ============================================================

\usepackage[utf8]{inputenc}
% moderncv loads hyperref itself in an \AtEndPreamble hook, so \hypersetup
% must go in an \AtEndPreamble of our own: on moderncv < 2.4 a top-level
% \usepackage{hyperref} clashes with the class's own
% \RequirePackage[unicode]{hyperref}. From 2.4.0 the class passes its options
% through \PassOptionsToPackage instead, which is what removes that clash.
\AtEndPreamble{\hypersetup{
    colorlinks=true,
    linkcolor=blue,
    filecolor=magenta,
    urlcolor=blue,
    pdftitle={[YOUR_NAME] - CV},
    % Keep pdfpagemode=UseNone: this block runs after moderncv's own
    % \AtEndPreamble (moderncv.cls sets pdfpagemode there), so a FullScreen
    % value here would win and open every CV in fullscreen presentation mode.
    pdfpagemode=UseNone,
}}
